# orders.{market_symbol}

## AsyncAPI Specification

```yaml
asyncapi: 2.6.0
info:
  title: orders.{market_symbol}
  version: subpackage_ordersMarketSymbol.ordersMarketSymbol
channels:
  /orders.{market_symbol}:
    parameters:
      market_symbol:
        description: Symbol of the market (e.g., BTC-PERP)
        schema:
          type: string

      operationId: orders-market-symbol-publish
      summary: subscribe
      message:
        name: subscribe
        title: subscribe
        payload:
          $ref: '#/components/schemas/OrdersMarketSymbolSubscribe'

    x-default: true

  schemas:

      type: string

      title: ChannelsOrdersMarketSymbolSubscribeInstruction
    ChannelsOrdersMarketSymbolSubscribeRequestInfo:
      type: object

          type: string

          type: string
          description: Type of request (MODIFY_ORDER)
        status:
          type: string
          description: Status of modify order request
      required:
        - id
        - message
        - request_type
        - status
      description: Additional request information for orders
      title: ChannelsOrdersMarketSymbolSubscribeRequestInfo

      type: string

        - BUY
        - SELL
      description: Order side

        - NEW
        - UNTRIGGERED
        - OPEN

      title: ChannelsOrdersMarketSymbolSubscribeStatus

      type: string

        market:
          type: string

        price:
          type: string
          description: Order price. 0 for MARKET orders

          type: integer

        type:
          $ref: '#/components/schemas/ChannelsOrdersMarketSymbolSubscribeType'

        - client_id
        - created_at
        - flags
        - id

        - market
```
